cmake_minimum_required(VERSION 3.20)
project(AssetBatchExperiment LANGUAGES C CXX)
# Isolated experiment: identical original converter objects serve both paths.
find_package(PNG REQUIRED)
find_package(ZLIB REQUIRED)
set(tools "${GAME_SOURCE}/tools")
set(gfx "${tools}/nitrogfx")
add_library(gfx_objects OBJECT "${gfx}/main.c" "${gfx}/convert_png.c" "${gfx}/gfx.c" "${gfx}/jasc_pal.c" "${gfx}/lz.c" "${gfx}/rl.c" "${gfx}/util.c" "${gfx}/font.c" "${gfx}/huff.c" "${gfx}/json.c" "${gfx}/cJSON.c")
set_source_files_properties("${gfx}/main.c" PROPERTIES COMPILE_DEFINITIONS main=hg_nitrogfx_main)
target_compile_features(gfx_objects PRIVATE c_std_17)
target_compile_definitions(gfx_objects PRIVATE PNG_SKIP_SETJMP_CHECK)
target_link_libraries(gfx_objects PUBLIC PNG::PNG ZLIB::ZLIB)
add_library(json_objects OBJECT "${tools}/jsonproc/jsonproc.cpp")
target_include_directories(json_objects PRIVATE "${tools}/jsonproc")
target_compile_definitions(json_objects PRIVATE main=hg_jsonproc_main)
target_compile_features(json_objects PRIVATE cxx_std_11)
add_executable(gfx_single single.c)
target_link_libraries(gfx_single PRIVATE gfx_objects)
add_executable(json_single single.cpp)
target_link_libraries(json_single PRIVATE json_objects)
add_executable(asset_worker worker.cpp)
target_compile_features(asset_worker PRIVATE cxx_std_17)
target_link_libraries(asset_worker PRIVATE gfx_objects json_objects)
file(GLOB arc_sources "${tools}/nitroarc/src/*.c")
add_executable(nitroarc "${tools}/nitroarc/lib/nitroarc.c" ${arc_sources})
target_include_directories(nitroarc PRIVATE "${tools}/nitroarc/lib/include")
add_executable(o2narc "${tools}/o2narc/o2narc.cpp" "${tools}/o2narc/Options.cpp" "${CMAKE_BINARY_DIR}/RelocElfReader.cpp")
target_include_directories(o2narc PRIVATE "${tools}/o2narc")
target_compile_features(o2narc PRIVATE cxx_std_17)
# Match ordinary production compilation; do not disable original assertions.
